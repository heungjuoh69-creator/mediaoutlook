rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isEditor() {
      return request.auth != null
        && request.auth.token.email == 'heungjuoh69@gmail.com'
        && request.auth.token.email_verified == true;
    }

    match /edits/{id} {
      allow read: if true;
      allow write: if isEditor();
    }
    match /admin/{id} {
      allow read, write: if isEditor();
    }
    match /views/{id} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['n'])
                    && request.resource.data.n == 1;
      allow update: if request.resource.data.keys().hasOnly(['n'])
                    && request.resource.data.n == resource.data.n + 1;
      allow delete: if isEditor();
    }
    match /archive/{id} {
      allow read: if true;
      allow write: if isEditor();
    }
    match /subscribers/{id} {
      allow create: if request.resource.data.email is string
                    && request.resource.data.name is string
                    && request.resource.data.org is string;
      allow read, update, delete: if isEditor();
    }
    match /leads/{id} {
      allow create: if request.resource.data.email is string
                    && request.resource.data.status == "신규";
      allow read, update, delete: if isEditor();
    }
    match /library/{id} {
      allow read: if true;
      allow write: if isEditor();
    }
    match /edposts/{id} {
      allow read: if true;
      allow write: if isEditor();
    }
    match /library_suggest/{id} {
      allow create: if true;
      allow read, update, delete: if isEditor();
    }
  }
}
